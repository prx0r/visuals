typescriptclass SwarmCoordinator extends DurableObject {
  totalCapital: number = 1000;  // $1000 total system capital
  
  // WEEKLY: Reallocate capital based on performance
  async reallocateCapital() {
    console.log('[SWARM] Reallocating capital based on pool performance...');
    
    // 1. Rank all pools by return
    const pools = await this.env.DB.prepare(`
      SELECT 
        p.id,
        p.name,
        p.objective_metric,
        p.current_capital,
        SUM(a.total_pnl_usd) as total_return,
        (SUM(a.total_pnl_usd) / p.starting_capital) as roi_pct,
        AVG(a.sharpe_ratio) as avg_sharpe
      FROM pools p
      JOIN agents a ON p.id = a.pool_id
      WHERE a.status = 'active'
      GROUP BY p.id
      ORDER BY roi_pct DESC
    `).all();
    
    // 2. Calculate new allocation (exponential: winners get more)
    const totalROI = pools.results.reduce((sum, p) => sum + Math.max(0, p.roi_pct), 0);
    
    const newAllocations = pools.results.map(pool => {
      // Winner bias: top pools get disproportionate capital
      const performanceWeight = Math.max(0, pool.roi_pct) / totalROI;
      const exponentialWeight = Math.pow(performanceWeight, 1.5);  // Amplify winners
      
      return {
        pool_id: pool.id,
        old_capital: pool.current_capital,
        new_capital: this.totalCapital * exponentialWeight,
        delta: (this.totalCapital * exponentialWeight) - pool.current_capital
      };
    });
    
    console.log('[SWARM] Capital reallocation:');
    newAllocations.forEach(a => {
      console.log(`  ${a.pool_id}: $${a.old_capital.toFixed(2)} → $${a.new_capital.toFixed(2)} (${a.delta > 0 ? '+' : ''}${a.delta.toFixed(2)})`);
    });
    
    // 3. Execute reallocation
    for (const allocation of newAllocations) {
      await this.transferCapital(allocation.pool_id, allocation.delta);
    }
    
    // 4. Notify pools of their new budget
    for (const allocation of newAllocations) {
      const overseer = this.getPoolOverseer(allocation.pool_id);
      await overseer.receiveCapitalUpdate(allocation.new_capital, allocation.delta);
    }
  }
  
  async transferCapital(poolId: string, delta: number) {
    if (delta > 0) {
      // Pool getting MORE capital
      console.log(`[SWARM] Allocating +$${delta.toFixed(2)} to ${poolId}`);
      
      await this.env.DB.prepare(`
        UPDATE pools SET current_capital = current_capital + ? WHERE id = ?
      `).bind(delta, poolId).run();
      
    } else if (delta < 0) {
      // Pool getting LESS capital (seize from underperformers)
      console.log(`[SWARM] Seizing $${Math.abs(delta).toFixed(2)} from ${poolId}`);
      
      // Kill agents proportionally to free up capital
      const overseer = this.getPoolOverseer(poolId);
      await overseer.reduceCapital(Math.abs(delta));
    }
  }
}

        symbol,
        SUM(CASE WHEN long_venue = 'lighter' THEN long_size * long_entry_price ELSE 0 END) as long_usd,

      FROM trades
      WHERE closed_at IS NULL
      GROUP BY symbol
    `).all();
    
    return {
      sol_net: positions.results.find(p => p.symbol === 'SOL-PERP')?.long_usd - positions.results.find(p => p.symbol === 'SOL-PERP')?.short_usd || 0,
      eth_net: positions.results.find(p => p.symbol === 'ETH-PERP')?.long_usd - positions.results.find(p => p.symbol === 'ETH-PERP')?.short_usd || 0,

      symbol,

      SUM(CASE WHEN short_venue = 'hyperliquid' THEN short_size * short_entry_price ELSE 0 END) as short_usd
    FROM trades
    WHERE closed_at IS NULL
    GROUP BY symbol
  `).all();
  
  const formatted = {};
  for (const exp of exposures.results) {
    formatted[exp.symbol] = {
      long: exp.long_usd,
      short: exp.short_usd,
      net: exp.long_usd - exp.short_usd
    };
  }
  
  return {
    exposures: formatted,
    timestamp: Date.now()
  };

      const agents = await this.env.DB.prepare(`
        SELECT * FROM agents
        WHERE pool_id = ? AND status = 'active'
        ORDER BY pool_score DESC
      `).bind(this.poolId).all();

      // Calculate cutoffs
      const promotionCutoff = Math.floor(agents.results.length * this.config.rules.promotion_threshold);
      const demotionCutoff = Math.floor(agents.results.length * (1 - this.config.rules.demotion_threshold));

      // Promote top performers
      for (let i = 0; i < promotionCutoff; i++) {
        const agent = agents.results[i];
        await this.promoteAgent(agent.id, agent.tier);
      }

      // Demote/kill bottom performers
      for (let i = demotionCutoff; i < agents.results.length; i++) {
        const agent = agents.results[i];

        if (agent.tier === 'bronze' && this.config.rules.death_penalty) {
          await this.killAgent(agent.id);
        } else {
          await this.demoteAgent(agent.id, agent.tier);
        }
      }

      console.log(`[POOL ${this.config.name}] ✅ Evaluation complete`);
    }

  // workers/MechanicalAgent.ts

  export class MechanicalAgent extends DurableObject {
    private agentId: string;
    private poolId: string;
    private rules: AgentRules;

    async alarm() {
      // Step 1: Get current environment from pool
      const environment = await this.getPoolEnvironment();

      // Step 2: Check death zone
      if (await this.isInDeathZone(environment)) {
        console.log(`[AGENT ${this.agentId}] In death zone, pausing`);
        return;
      }

      // Step 3: Check mechanical entry criteria
      const opportunity = await this.checkOpportunity(environment);

      if (opportunity) {
        await this.executeTrade(opportunity, environment);
      }

      // Step 4: Check exits for open positions
      await this.checkExits(environment);

      // Step 5: Schedule next alarm
      await this.ctx.storage.setAlarm(Date.now() + 3600000);  // 1 hour
    }

    async getPoolEnvironment(): Promise<EnvironmentalConditions> {
      // Fetch from pool overseer
      const pool = await this.env.DB.prepare(`
        SELECT durable_object_id FROM pools WHERE id = ?
      `).bind(this.poolId).first();

      const doId = this.env.POOL_OVERSEER.idFromString(pool.durable_object_id);
      const overseer = this.env.POOL_OVERSEER.get(doId);

      const response = await overseer.fetch(new Request('https://fake/environment'));
      return await response.json();
    }

    async checkOpportunity(env: EnvironmentalConditions): Promise<Opportunity | null> {
      const market = await this.getMarketData();

      // MECHANICAL CHECKS (no AI!)
      const checks = [
        // Pool-set thresholds
        market.spread_bps >= env.min_spread_threshold,
        market.spread_bps <= env.max_spread_threshold,

        // Agent-specific rules
        this.rules.strategy_type === 'whale_follower' ? market.whale_activity : true,
        this.rules.strategy_type === 'low_liquidity' ? market.oi < 1000000 : true,

        // Time constraints
        env.allowed_hours.includes(new Date().getUTCHours()),

        // Cooldown
        Date.now() - this.lastTradeTime > env.trade_cooldown_minutes * 60 * 1000,

        // Position limits
        this.openPositions < env.max_open_positions
      ];

      return checks.every(c => c) ? market : null;
    }
  }
