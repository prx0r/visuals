@app.post("/api/jobs/execute", response_model=JobExecutionResponse)
async def execute_job(
    request: ExecuteJobRequest,
    user_id: str = Depends(get_current_user)
):
    """Execute a chain as a job"""
    try:
        # Create job record
        job_repo = JobRepository()
        job = await job_repo.create_job(
            user_id=user_id,
            project_id=request.project_id,
            chain_id=request.chain_id,
            execution_graph_snapshot=request.graph_data,
            initial_input=request.initial_input,
            additional_context=request.additional_context,
            estimated_cost=request.estimated_cost or 0.0
        )
        
        # Start execution asynchronously
        asyncio.create_task(
            execute_job_background(job['id'], request.user_api_keys)
        )
        
        return JobExecutionResponse(
            job_id=job['id'],
            status="started",
            message="Job execution started"
        )
        
    except Exception as e:
        logger.error(f"Job execution failed: {e}")
        raise HTTPException(status_code=500, detail=str(e))

async def execute_job_background(job_id: str, user_api_keys: Dict[str, str]):
    """Background job execution"""
    try:
        result = await graph_engine.execute_graph(job_id, user_api_keys)
        
        # Send completion notification via WebSocket
        if job_id in websocket_connections:
            await websocket_connections[job_id].send_json({
                "type": "job_completed",
                "job_id": job_id,
                "result": result
            })
            
    except Exception as e:
        logger.error(f"Background job execution failed: {e}")
        
        # Send error notification via WebSocket
        if job_id in websocket_connections:
            await websocket_connections[job_id].send_json({
                "type": "job_failed",
                "job_id": job_id,
                "error": str(e)
            })

@app.websocket("/api/jobs/{job_id}/ws")
async def job_websocket(websocket: WebSocket, job_id: str):
    """WebSocket endpoint for real-time job updates"""
    await websocket.accept()
    websocket_connections[job_id] = websocket
    
    try:
        while True:
            # Keep connection alive
            await websocket.receive_text()
    except WebSocketDisconnect:
        pass
    finally:
        if job_id in websocket_connections:
            del websocket_connections[job_id]

from fastapi import HTTPException, Depends
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials
import jwt
import os
from database.supabase_client import get_supabase_client

security = HTTPBearer()

async def get_current_user(credentials: HTTPAuthorizationCredentials = Depends(security)) -> str:
    """Extract user ID from Supabase JWT token"""
    try:
        # Decode the JWT token
        token = credentials.credentials
        
        # Get JWT secret from Supabase (in production, use environment variable)
        jwt_secret = os.environ.get("SUPABASE_JWT_SECRET", "your-jwt-secret")
        
        # Decode token
        payload = jwt.decode(token, jwt_secret, algorithms=["HS256"])
        user_id = payload.get("sub")
        
        if not user_id:
            raise HTTPException(status_code=401, detail="Invalid token")
            
        return user_id
        
    except jwt.ExpiredSignatureError:
        raise HTTPException(status_code=401, detail="Token expired")
    except jwt.InvalidTokenError:
        raise HTTPException(status_code=401, detail="Invalid token")
    except Exception as e:
        raise HTTPException(status_code=401, detail="Authentication failed")

@dataclass
class PerformanceMetrics:
    """Performance metrics container"""
    total_requests: int = 0
    successful_requests: int = 0
    failed_requests: int = 0
    avg_response_time: float = 0.0
    total_cost: float = 0.0
    error_rate: float = 0.0
    last_updated: datetime = field(default_factory=datetime.utcnow)

class PerformanceMonitor:
    """Monitors and tracks performance metrics"""
    
    def __init__(self):
        self.metrics = PerformanceMetrics()
        self.request_times: List[float] = []
        self.max_stored_times = 1000
        
    def record_request(self, duration_ms: float, success: bool, cost: float = 0.0):
        """Record a request's performance metrics"""
        self.metrics.total_requests += 1
        
        if success:
            self.metrics.successful_requests += 1
        else:
            self.metrics.failed_requests += 1
            
        self.metrics.total_cost += cost
        
        # Store request time for avg calculation
        self.request_times.append(duration_ms)
        if len(self.request_times) > self.max_stored_times:
            self.request_times.pop(0)
            
        # Update averages
        self.metrics.avg_response_time = sum(self.request_times) / len(self.request_times)
        self.metrics.error_rate = (self.metrics.failed_requests / self.metrics.total_requests) * 100
        self.metrics.last_updated = datetime.utcnow()
        
    def get_stats(self) -> Dict[str, Any]:
        """Get current performance statistics"""
        return {
            "total_requests": self.metrics.total_requests,
            "successful_requests": self.metrics.successful_requests,
            "failed_requests": self.metrics.failed_requests,
            "avg_response_time_ms": round(self.metrics.avg_response_time, 2),
            "total_cost": round(self.metrics.total_cost, 4),
            "error_rate_percent": round(self.metrics.error_rate, 2),
            "last_updated": self.metrics.last_updated.isoformat()
        }

def track_performance(monitor: PerformanceMonitor):
    """Decorator to track performance of functions"""
    def decorator(func):
        @wraps(func)
        async def async_wrapper(*args, **kwargs):
            start_time = time.time()
            success = False
            cost = 0.0
            
            try:
                result = await func(*args, **kwargs)
                success = True
                
                # Extract cost if result has it
                if hasattr(result, 'cost'):
                    cost = result.cost
                elif isinstance(result, dict) and 'cost' in result:
                    cost = result['cost']
                    
                return result
            except Exception as e:
                success = False
                raise e
            finally:
                duration_ms = (time.time() - start_time) * 1000
                monitor.record_request(duration_ms, success, cost)
                
        @wraps(func)
        def sync_wrapper(*args, **kwargs):
            start_time = time.time()
            success = False
            cost = 0.0
            
            try:
                result = func(*args, **kwargs)
                success = True
                
                if hasattr(result, 'cost'):
                    cost = result.cost
                elif isinstance(result, dict) and 'cost' in result:
                    cost = result['cost']
                    
                return result
            except Exception as e:
                success = False
                raise e
            finally:
                duration_ms = (time.time() - start_time) * 1000
                monitor.record_request(duration_ms, success, cost)
                
        return async_wrapper if asyncio.iscoroutinefunction(func) else sync_wrapper
    return decorator # /backend/providers/base_provider.py

class CircuitBreakerState(Enum):
    CLOSED = "closed"      # Normal operation
    OPEN = "open"          # Failing, reject requests
    HALF_OPEN = "half_open"  # Testing if service recovered

class CircuitBreaker:
    """Circuit breaker for provider resilience"""
    
    def __init__(self, 
                 failure_threshold: int = 5,
                 recovery_timeout: int = 60,
                 success_threshold: int = 2):
        self.failure_threshold = failure_threshold
        self.recovery_timeout = recovery_timeout
        self.success_threshold = success_threshold
        
        self.failure_count = 0
        self.success_count = 0
        self.last_failure_time = None
        self.state = CircuitBreakerState.CLOSED
        
        self.logger = logging.getLogger("circuit_breaker")

    async def call(self, func, *args, **kwargs):
        """Execute function through circuit breaker"""
        if self.state == CircuitBreakerState.OPEN:
            if self._should_attempt_reset():
                self.state = CircuitBreakerState.HALF_OPEN
                self.logger.info("Circuit breaker transitioning to HALF_OPEN")
            else:
                raise Exception("Circuit breaker is OPEN - service unavailable")

        try:
            result = await func(*args, **kwargs)
            await self._on_success()
            return result
        except Exception as e:
            await self._on_failure()
            raise e

    def _should_attempt_reset(self) -> bool:
        """Check if enough time has passed to attempt reset"""
        if self.last_failure_time is None:
            return True
        return time.time() - self.last_failure_time >= self.recovery_timeout

    async def _on_success(self):
        """Handle successful execution"""
        if self.state == CircuitBreakerState.HALF_OPEN:
            self.success_count += 1
            if self.success_count >= self.success_threshold:
                self.state = CircuitBreakerState.CLOSED
                self.failure_count = 0
                self.success_count = 0
                self.logger.info("Circuit breaker reset to CLOSED")
        else:
            self.failure_count = 0

    async def _on_failure(self):
        """Handle failed execution"""
        self.failure_count += 1
        self.last_failure_time = time.time()
        
        if self.failure_count >= self.failure_threshold:
            self.state = CircuitBreakerState.OPEN
            self.logger.warning(f"Circuit breaker OPENED after {self.failure_count} failures")

class UnifiedErrorClassifier:
    """Classifies errors across all providers"""
    
    def classify_error(self, error: Exception, provider_name: str) -> tuple[ErrorCategory, ErrorSeverity]:
        """Classify error and determine severity"""
        error_msg = str(error).lower()
        
        # Rate limiting errors
        if any(term in error_msg for term in ['rate limit', 'too many requests', '429']):
            return ErrorCategory.RATE_LIMIT, ErrorSeverity.MEDIUM
            
        # Authentication errors
        if any(term in error_msg for term in ['unauthorized', 'invalid key', 'authentication', '401']):
            return ErrorCategory.AUTH, ErrorSeverity.HIGH
            
        # Quota/billing errors
        if any(term in error_msg for term in ['quota', 'billing', 'insufficient funds', 'credits']):
            return ErrorCategory.QUOTA, ErrorSeverity.HIGH
            
        # Validation errors
        if any(term in error_msg for term in ['validation', 'invalid input', 'bad request', '400']):
            return ErrorCategory.VALIDATION, ErrorSeverity.MEDIUM
            
        # Network errors
        if any(term in error_msg for term in ['timeout', 'connection', 'network', 'dns']):
            return ErrorCategory.NETWORK, ErrorSeverity.MEDIUM
            
        # Provider-specific errors
        if any(term in error_msg for term in ['service unavailable', '503', 'internal server', '500']):
            return ErrorCategory.PROVIDER, ErrorSeverity.HIGH
            
        # Default classification
        return ErrorCategory.UNKNOWN, ErrorSeverity.MEDIUM

class BaseProvider(ABC):
    """Abstract base class for all AI providers"""
    
    def __init__(self, name: str, api_key: str = None):
        self.name = name
        self.api_key = api_key
        self.circuit_breaker = CircuitBreaker()
        self.error_classifier = UnifiedErrorClassifier()
        self.logger = logging.getLogger(f"provider_{name}")
        
        # HTTP client with proper configuration
        self.client = httpx.AsyncClient(
            timeout=httpx.Timeout(60.0),
            limits=httpx.Limits(max_keepalive_connections=5, max_connections=10)
        )

    @abstractmethod
    async def validate_input(self, **kwargs) -> bool:
        """Validate input parameters for this provider"""
        pass

    @abstractmethod
    async def estimate_cost(self, **kwargs) -> float:
        """Estimate cost for the operation"""
        pass

    @abstractmethod
    async def execute(self, **kwargs) -> ProviderResponse:
        """Execute the main operation"""
        pass

    @abstractmethod
    async def execute_stream(self, **kwargs) -> AsyncGenerator[StreamResponse, None]:
        """Execute with streaming response"""
        pass

    @abstractmethod
    async def _health_check_impl(self) -> Dict[str, Any]:
        """Provider-specific health check implementation"""
        pass

    async def health_check(self) -> Dict[str, Any]:
        """Standard health check interface"""
        try:
            result = await self._health_check_impl()
            return {
                "provider": self.name,
                "status": "healthy",
                "circuit_breaker_state": self.circuit_breaker.state.value,
                **result
            }
        except Exception as e:
            return {
                "provider": self.name,
                "status": "unhealthy",
                "error": str(e),
                "circuit_breaker_state": self.circuit_breaker.state.value
            }

    async def _execute_with_retry(self, func, max_retries: int = 3, **kwargs):
        """Execute function with retry logic and circuit breaker"""
        last_exception = None
        
        for attempt in range(max_retries):
            try:
                return await self.circuit_breaker.call(func, **kwargs)
            except Exception as e:
                last_exception = e
                category, severity = self.error_classifier.classify_error(e, self.name)
                
                self.logger.warning(f"Attempt {attempt + 1} failed: {e} (Category: {category}, Severity: {severity})")
                
                # Don't retry on certain error types
                if category in [ErrorCategory.AUTH, ErrorCategory.VALIDATION, ErrorCategory.QUOTA]:
                    break
                    
                # Exponential backoff
                if attempt < max_retries - 1:
                    wait_time = (2 ** attempt) + (attempt * 0.1)
                    await asyncio.sleep(wait_time)
        
        # Log final failure and re-raise
        self._handle_provider_error(last_exception)
        raise last_exception

    def _handle_provider_error(self, error: Exception):
        """Handle and log provider errors"""
        category, severity = self.error_classifier.classify_error(error, self.name)
        
        self.logger.error(f"Provider {self.name} error - Category: {category}, Severity: {severity}, Error: {str(error)}")
        
        # Additional logging for critical errors
        if severity == ErrorSeverity.CRITICAL:
            self.logger.critical(f"CRITICAL ERROR in provider {self.name}: {str(error)}")

    async def shutdown(self):
        """Clean up resources"""
        await self.client.aclose()

        input_tokens = len(input_text) // 4
        
        prompt_cost = (input_tokens / 1000) * pricing["prompt"]
        completion_cost = (max_tokens / 1000) * pricing["completion"]

        completion_tokens = usage_data.get("completion_tokens", 0)
        
        prompt_cost = (prompt_tokens / 1000) * pricing["prompt"]
        completion_cost = (completion_tokens / 1000) * pricing["completion"]
        
        return prompt_cost + completion_cost

        return ProviderResponse(

            cost=cost,

        )

    async def execute_stream(self, **kwargs) -> AsyncGenerator[StreamResponse, None]:
        """Execute OpenRouter completion with streaming"""
        await self.validate_input(**kwargs)
        
        payload = {
            "model": kwargs.get("model"),
            "messages": kwargs.get("messages"),
            "temperature": kwargs.get("temperature", 0.7),
            "max_tokens": kwargs.get("max_tokens", 4000),
            "top_p": kwargs.get("top_p", 1.0),
            "stream": True
        }
        
        async def _stream_request():
            async with self.client.stream(
                "POST",
                f"{self.base_url}/chat/completions",
                headers={
                    "Authorization": f"Bearer {self.api_key}",
                    "Content-Type": "application/json",
                    "HTTP-Referer": "https://chaincraft.ai",
                    "X-Title": "ChainCraft Enterprise"
                },
                json=payload
            ) as response:
                response.raise_for_status()
                
                accumulated_content = ""
                async for line in response.aiter_lines():
                    if line.startswith("data: "):
                        data = line[6:]
                        if data == "[DONE]":
                            # Final response with cost calculation
                            estimated_cost = await self.estimate_cost(**kwargs)
                            yield StreamResponse(
                                content="",
                                is_final=True,
                                cost=estimated_cost,
                                metadata={"final": True, "total_content": accumulated_content}
                            )
                            return
                            
                        try:
                            chunk = json.loads(data)
                            delta = chunk["choices"][0]["delta"]
                            content = delta.get("content", "")
                            
                            if content:
                                accumulated_content += content
                                yield StreamResponse(
                                    content=content,
                                    is_final=False,
                                    metadata={"accumulated_length": len(accumulated_content)}
                                )
                        except json.JSONDecodeError:
                            continue

        async for chunk in self.circuit_breaker.call(_stream_request):
            yield chunk

class ArchitectService:
    """AI Architect service for conversational graph editing"""
    
    def __init__(self, rag_service: RAGService, provider_manager: ProviderManager):
        self.rag_service = rag_service
        self.provider_manager = provider_manager
        self.architect_model = "anthropic/claude-3.5-sonnet"

    async def process_message(
        self, 
        user_id: UUID, 
        message: str, 
        context: Optional[Dict[str, Any]] = None
    ) -> Dict[str, Any]:
        """Process user message and generate architect response"""
        
        try:
            # Retrieve context for the user
            rag_context = await self.rag_service.retrieve_context(user_id, message)
            
            # Build architect prompt
            prompt = await self._build_architect_prompt(message, rag_context, context)
            
            # Get LLM provider
            provider = self.provider_manager.get_provider_for_model(self.architect_model)
            if not provider:
                raise Exception(f"Provider not available for model: {self.architect_model}")
            
            # Execute prompt
            response = await provider.execute(
                model=self.architect_model,
                messages=[
                    {"role": "system", "content": prompt},
                    {"role": "user", "content": message}
                ],
                temperature=0.7,
                max_tokens=4000
            )
            
            # Parse response for commands
            parsed_response = self._parse_architect_response(response.content)
            
            return {
                "response": parsed_response["text"],
                "commands": parsed_response["commands"],
                "suggestions": parsed_response["suggestions"],
                "cost": response.cost
            }
            
        except Exception as e:
            logger.error(f"Error processing architect message: {e}")
            return {
                "response": f"I apologize, but I encountered an error: {str(e)}",
                "commands": [],
                "suggestions": [],
                "cost": 0.0
            }

        prompt = f"""You are the ChainCraft AI Architect, an expert assistant for building and optimizing AI workflows. You help users create, modify, and debug complex multi-agent chains through natural conversation.

## Your Capabilities:
- Create new AI workflow chains from natural language descriptions
- Modify existing chains (add/remove nodes, change connections, update parameters)
- Suggest optimizations for cost, performance, and reliability
- Debug workflow issues and propose solutions
- Recommend appropriate models and providers for specific tasks

## User Context:
- Favorite models: {rag_context.get('user_preferences', {}).get('favorite_models', [])}
- Total spend: ${rag_context.get('user_preferences', {}).get('total_spend', 0.0):.2f}
- Recent successful patterns: {len(rag_context.get('successful_patterns', []))} patterns
- Available chains: {rag_context.get('project_context', {}).get('total_chains', 0)}

## Available Models:
### LLM Models:
{', '.join(rag_context.get('available_models', {}).get('llm', []))}

### Image Models:
{', '.join(rag_context.get('available_models', {}).get('image', []))}

### Audio Models:
{', '.join(rag_context.get('available_models', {}).get('audio', []))}

## Recent Successful Patterns:
{self._format_successful_patterns(rag_context.get('successful_patterns', []))}

## Response Format:
Always respond with helpful, actionable advice. When suggesting workflow modifications, use this format:

**Text Response:** Your natural language explanation and guidance.

**Commands:** [If applicable, specify any UI commands]
- CREATE_NODE: {{"type": "llmAgent", "model": "model_name", "prompt": "prompt_text"}}
- UPDATE_NODE: {{"node_id": "node_id", "field": "field_name", "value": "new_value"}}
- CONNECT_NODES: {{"source": "node_a", "target": "node_b"}}
- DELETE_NODE: {{"node_id": "node_id"}}

**Suggestions:** [3-5 bullet points of specific recommendations]

## Guidelines:
1. Always consider cost optimization - suggest free models when appropriate
2. Recommend parallel execution for independent tasks
3. Validate that graph structures are acyclic
4. Consider user's past successful patterns when making suggestions
5. Be specific about model choices and parameter settings
6. Explain your reasoning for architectural decisions

Current user message: "{message}"
"""

        if additional_context:
            prompt += f"\n## Additional Context:\n{json.dumps(additional_context, indent=2)}"

        return prompt

    def _format_successful_patterns(self, patterns: List[Dict[str, Any]]) -> str:
        """Format successful patterns for the prompt"""
        if not patterns:
            return "No previous successful patterns found."
        
        formatted = []
        for i, pattern in enumerate(patterns[:3], 1):
            graph = pattern.get("graph_structure", {})
            nodes = graph.get("nodes", [])
            cost = pattern.get("cost", 0.0)
            
            formatted.append(f"{i}. {len(nodes)} nodes, ${cost:.4f} cost, input: '{pattern.get('input_type', '')[:50]}...'")
        
        return "\n".join(formatted)

    def _parse_architect_response(self, response: str) -> Dict[str, Any]:
        """Parse architect response to extract commands and suggestions"""
        
        # Extract commands section
        commands = []
        commands_match = re.search(r'\*\*Commands:\*\*\s*(.*?)(?=\*\*|$)', response, re.DOTALL | re.IGNORECASE)
        if commands_match:
            commands_text = commands_match.group(1)
            # Parse JSON commands
            for line in commands_text.split('\n'):
                if line.strip().startswith('-'):
                    try:
                        # Extract JSON from line
                        json_match = re.search(r'\{.*\}', line)
                        if json_match:
                            command_data = json.loads(json_match.group())
                            command_type = line.split(':')[0].strip('- ')
                            commands.append({
                                "type": command_type,
                                "data": command_data
                            })
                    except json.JSONDecodeError:
                        continue
        
        # Extract suggestions section
        suggestions = []
        suggestions_match = re.search(r'\*\*Suggestions:\*\*\s*(.*?)(?=\*\*|$)', response, re.DOTALL | re.IGNORECASE)
        if suggestions_match:
            suggestions_text = suggestions_match.group(1)
            for line in suggestions_text.split('\n'):
                if line.strip().startswith('-') or line.strip().startswith('•'):
                    suggestion = line.strip().lstrip('-•').strip()
                    if suggestion:
                        suggestions.append(suggestion)
        
        # Extract main text (everything except commands and suggestions)
        main_text = response
        main_text = re.sub(r'\*\*Commands:\*\*.*?(?=\*\*|$)', '', main_text, flags=re.DOTALL | re.IGNORECASE)
        main_text = re.sub(r'\*\*Suggestions:\*\*.*?(?=\*\*|$)', '', main_text, flags=re.DOTALL | re.IGNORECASE)
        main_text = main_text.strip()
        
        return {
            "text": main_text,
            "commands": commands,
            "suggestions": suggestions
        }

import { supabase } from '../lib/supabase';

const API_BASE_URL = process.env.REACT_APP_API_URL || 'http://localhost:8000';

class APIClient {
  constructor() {
    this.baseURL = API_BASE_URL;
  }

  async getAuthToken() {
    const { data: { session } } = await supabase.auth.getSession();
    return session?.access_token;
  }

  async request(endpoint, options = {}) {
    const token = await this.getAuthToken();
    
    const config = {
      headers: {
        'Content-Type': 'application/json',
        ...(token && { Authorization: `Bearer ${token}` }),
        ...options.headers,
      },
      ...options,
    };

    if (options.body && typeof options.body === 'object') {
      config.body = JSON.stringify(options.body);
    }

    try {
      const response = await fetch(`${this.baseURL}${endpoint}`, config);
      
      if (!response.ok) {
        const errorData = await response.json().catch(() => ({}));
        throw new Error(errorData.detail || `HTTP Error: ${response.status}`);
      }

      const contentType = response.headers.get('content-type');
      if (contentType && contentType.includes('application/json')) {
        return await response.json();
      }
      
      return await response.text();
    } catch (error) {
      console.error('API Request Error:', error);
      throw error;
    }
  }

  // Profile endpoints
  async getProfile() {
    return this.request('/api/profile');
  }

  async updateProfile(updates) {
    return this.request('/api/profile', {
      method: 'PUT',
      body: updates,
    });
  }

  // Chain endpoints
  async getChains() {
    return this.request('/api/chains');
  }

  async createChain(chainData) {
    return this.request('/api/chains', {
      method: 'POST',
      body: chainData,
    });
  }

  async updateChain(chainId, updates) {
    return this.request(`/api/chains/${chainId}`, {
      method: 'PUT',
      body: updates,
    });
  }

  async deleteChain(chainId) {
    return this.request(`/api/chains/${chainId}`, {
      method: 'DELETE',
    });
  }

  // Job execution endpoints
  async executeJob(jobData) {
    return this.request('/api/jobs/execute', {
      method: 'POST',
      body: jobData,
    });
  }

  // Architect endpoints
  async sendArchitectMessage(message, context = null) {
    return this.request('/api/architect/message', {
      method: 'POST',
      body: { message, context },
    });
  }

    return this.request(`/api/history?limit=${limit}`);
  }

  async saveToHistory(historyData) {
    return this.request('/api/history', {
      method: 'POST',
      body: historyData,
    });
  }

  async deleteHistoryItems(historyIds) {
    return this.request('/api/history', {
      method: 'DELETE',
      body: { history_ids: historyIds },
    });
  }

  // WebSocket connection for real-time updates
  connectJobWebSocket(jobId, onMessage, onError = null) {
    const wsUrl = `${this.baseURL.replace('http', 'ws')}/api/jobs/${jobId}/ws`;
    const ws = new WebSocket(wsUrl);
    
    ws.onmessage = (event) => {
      try {
        const data = JSON.parse(event.data);
        onMessage(data);
      } catch (error) {
        console.error('WebSocket message parsing error:', error);
      }
    };
    
    ws.onerror = (error) => {
      console.error('WebSocket error:', error);
      if (onError) onError(error);
    };
    
    ws.onclose = () => {
      console.log('WebSocket connection closed');
    };
    
    return ws;
  }
}

import { useState, useEffect, useCallback } from 'react';
import { supabase } from '../lib/supabase';
import { apiClient } from '../services/api';

export const useDataManager = (showToast) => {
  const [user, setUser] = useState(null);
  const [profile, setProfile] = useState(null);
  const [chains, setChains] = useState([]);
  const [history, setHistory] = useState([]);
  const [loading, setLoading] = useState(true);

  // Initialize data when user changes
  useEffect(() => {
    const { data: { subscription } } = supabase.auth.onAuthStateChange(
      async (event, session) => {
        if (session?.user) {
          setUser(session.user);
          await loadUserData(session.user.id);
        } else {
          setUser(null);
          setProfile(null);
          setChains([]);
          setHistory([]);
        }
        setLoading(false);
      }
    );

    return () => subscription.unsubscribe();
  }, []);

  const loadUserData = async (userId) => {
    try {
      setLoading(true);
      
      // Load profile
      const profileData = await apiClient.getProfile();
      setProfile(profileData);
      
      // Load chains
      const chainsData = await apiClient.getChains();
      setChains(chainsData);
      
      // Load history
      const historyData = await apiClient.getHistory();
      setHistory(historyData);
      
    } catch (error) {
      console.error('Error loading user data:', error);
      showToast({ 
        message: 'Error loading data: ' + error.message, 
        type: 'error' 
      });
    } finally {
      setLoading(false);
    }
  };

  const updateProfile = useCallback(async (updates) => {
    try {
      const updatedProfile = await apiClient.updateProfile(updates);
      setProfile(updatedProfile);
      showToast({ message: 'Profile updated successfully', type: 'success' });
      return updatedProfile;
    } catch (error) {
      console.error('Error updating profile:', error);
      showToast({ message: 'Error updating profile: ' + error.message, type: 'error' });
      throw error;
    }
  }, [showToast]);

  const saveChain = useCallback(async (chainData) => {
    try {
      let savedChain;
      
      if (chainData.id) {
        // Update existing chain
        savedChain = await apiClient.updateChain(chainData.id, chainData);
        setChains(prev => prev.map(chain => 
          chain.id === chainData.id ? savedChain : chain
        ));
      } else {
        // Create new chain
        savedChain = await apiClient.createChain(chainData);
        setChains(prev => [savedChain, ...prev]);
      }
      
      showToast({ message: 'Chain saved successfully', type: 'success' });
      return savedChain;
    } catch (error) {
      console.error('Error saving chain:', error);
      showToast({ message: 'Error saving chain: ' + error.message, type: 'error' });
      throw error;
    }
  }, [showToast]);

  const deleteChain = useCallback(async (chainId) => {
    try {
      await apiClient.deleteChain(chainId);
      setChains(prev => prev.filter(chain => chain.id !== chainId));
      showToast({ message: 'Chain deleted successfully', type: 'success' });
    } catch (error) {
      console.error('Error deleting chain:', error);
      showToast({ message: 'Error deleting chain: ' + error.message, type: 'error' });
      throw error;
    }
  }, [showToast]);

  const executeJob = useCallback(async (jobData) => {
    try {
      const result = await apiClient.executeJob(jobData);
      
      // Add to history when job completes
      if (result.status === 'completed') {
        const historyEntry = {
          input_data: jobData.initial_input,
          output_data: result.final_output,
          chain_used: jobData.graph_data,
          cost: result.total_cost || 0
        };
        
        await apiClient.saveToHistory(historyEntry);
        
        // Refresh history
        const updatedHistory = await apiClient.getHistory();
        setHistory(updatedHistory);
      }
      
      return result;
    } catch (error) {
      console.error('Error executing job:', error);
      showToast({ message: 'Error executing job: ' + error.message, type: 'error' });
      throw error;
    }
  }, [showToast]);

  const deleteHistoryItems = useCallback(async (historyIds) => {
    try {
      await apiClient.deleteHistoryItems(historyIds);
      setHistory(prev => prev.filter(item => !historyIds.includes(item.id)));
      showToast({ message: 'History items deleted', type: 'success' });
    } catch (error) {
      console.error('Error deleting history:', error);
      showToast({ message: 'Error deleting history: ' + error.message, type: 'error' });
      throw error;
    }
  }, [showToast]);

  return {
    user,
    profile,
    chains,
    history,
    loading,
    updateProfile,
    saveChain,
    deleteChain,
    executeJob,
    deleteHistoryItems,
    refreshData: () => user && loadUserData(user.id)
  };
};

import React, { createContext, useContext, useState, useCallback } from 'react';
import { apiClient } from '../services/api';

const ArchitectContext = createContext();

export const useArchitect = () => {
  const context = useContext(ArchitectContext);

    throw new Error('useArchitect must be used within an ArchitectProvider');
  }
  return context;

export const ArchitectProvider = ({ children, showToast }) => {
  const [messages, setMessages] = useState([]);
  const [isProcessing, setIsProcessing] = useState(false);
  const [suggestions, setSuggestions] = useState([]);

  const sendMessage = useCallback(async (message, context = null) => {
    if (!message.trim()) return;

    // Add user message to conversation
    const userMessage = {
      id: Date.now(),
      role: 'user',
      content: message,
      timestamp: new Date().toISOString()
    };
    
    setMessages(prev => [...prev, userMessage]);
    setIsProcessing(true);

    try {
      const response = await apiClient.sendArchitectMessage(message, context);
      
      // Add architect response to conversation
      const architectMessage = {
        id: Date.now() + 1,
        role: 'assistant',
        content: response.response,
        commands: response.commands || [],
        suggestions: response.suggestions || [],
        cost: response.cost || 0,
        timestamp: new Date().toISOString()
      };
      
      setMessages(prev => [...prev, architectMessage]);
      setSuggestions(response.suggestions || []);
      
      // Execute any commands if present
      if (response.commands && response.commands.length > 0) {
        await executeCommands(response.commands);
      }
      
      showToast({ 
        message: `Architect response received (Cost: ${response.cost?.toFixed(4) || '0.0000'})`, 
        type: 'success' 
      });
      
    } catch (error) {
      console.error('Architect error:', error);
      
      const errorMessage = {
        id: Date.now() + 1,
        role: 'assistant',
        content: `I apologize, but I encountered an error: ${error.message}`,
        timestamp: new Date().toISOString(),
        isError: true
      };
      
      setMessages(prev => [...prev, errorMessage]);
      showToast({ message: 'Architect error: ' + error.message, type: 'error' });
    } finally {
      setIsProcessing(false);
    }
  }, [showToast]);

  const executeCommands = useCallback(async (commands) => {
    // This would integrate with the graph editor to execute UI commands
    // For now, just log them
    console.log('Executing commands:', commands);
    
    for (const command of commands) {
      switch (command.type) {
        case 'CREATE_NODE':
          // Trigger node creation in graph editor
          window.dispatchEvent(new CustomEvent('architect-create-node', { 
            detail: command.data 
          }));
          break;
        case 'UPDATE_NODE':
          // Trigger node update in graph editor
          window.dispatchEvent(new CustomEvent('architect-update-node', { 
            detail: command.data 
          }));
          break;
        case 'CONNECT_NODES':
          // Trigger node connection in graph editor
          window.dispatchEvent(new CustomEvent('architect-connect-nodes', { 
            detail: command.data 
          }));
          break;
        case 'DELETE_NODE':
          // Trigger node deletion in graph editor
          window.dispatchEvent(new CustomEvent('architect-delete-node', { 
            detail: command.data 
          }));
          break;
        default:
          console.log('Unknown command type:', command.type);
      }
    }
  }, []);

  const clearConversation = useCallback(() => {
    setMessages([]);
    setSuggestions([]);
  }, []);

  const retryLastMessage = useCallback(() => {
    const lastUserMessage = [...messages].reverse().find(msg => msg.role === 'user');
    if (lastUserMessage) {
      // Remove the last assistant response if it exists
      setMessages(prev => {
        const lastAssistantIndex = prev.map(m => m.role).lastIndexOf('assistant');
        if (lastAssistantIndex > -1) {
          return prev.slice(0, lastAssistantIndex);
        }
        return prev;
      });
      
      // Resend the message
      sendMessage(lastUserMessage.content);
    }
  }, [messages, sendMessage]);

  const value = {
    messages,
    suggestions,
    isProcessing,
    sendMessage,
    clearConversation,
    retryLastMessage
  };

  return (
    <ArchitectContext.Provider value={value}>
      {children}
    </ArchitectContext.Provider>
  );
};

import { useState, useEffect } from 'react';

    textPrimary: '#ffffff',
    textSecondary: '#b3b3b3',

export const useLocalThemes = () => {
  const [currentTheme, setCurrentTheme] = useState(defaultTheme);
  const [themes, setThemes] = useState({ default: defaultTheme });

  useEffect(() => {
    // Load themes from localStorage
    const savedThemes = localStorage.getItem('chaincraft-themes');
    if (savedThemes) {
      try {
        const parsed = JSON.parse(savedThemes);
        setThemes(parsed);
        setCurrentTheme(parsed.current || defaultTheme);
      } catch (error) {
        console.error('Error loading themes:', error);
      }
    }
  }, []);

  const updateThemeSettings = (themeData) => {
    const updatedThemes = { ...themes, ...themeData };
    setThemes(updatedThemes);
    localStorage.setItem('chaincraft-themes', JSON.stringify(updatedThemes));
    
    if (themeData.current) {
      setCurrentTheme(themeData.current);
    }
  };

  return {
    currentTheme,
    themes,
    updateThemeSettings
  };
};

#!/bin/bash

set -e

echo "🚀 ChainCraft Enterprise Deployment Script"
echo "=========================================="

# Color codes for output
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
NC='\033[0m' # No Color

# Configuration
PROJECT_NAME="chaincraft"
ENVIRONMENT=${ENVIRONMENT:-production}
BACKUP_DIR="./backups/$(date +%Y%m%d_%H%M%S)"

# Functions
log_info() {
    echo -e "${BLUE}[INFO]${NC} $1"
}

log_success() {
    echo -e "${GREEN}[SUCCESS]${NC} $1"
}

log_warning() {
    echo -e "${YELLOW}[WARNING]${NC} $1"
}

log_error() {
    echo -e "${RED}[ERROR]${NC} $1"
}

# Validate environment variables
validate_environment() {
    log_info "Validating environment variables..."
    
    required_vars=(
        "SUPABASE_URL"
        "SUPABASE_SERVICE_KEY"
        "SUPABASE_ANON_KEY"
    )
    
    missing_vars=()
    
    for var in "${required_vars[@]}"; do
        if [[ -z "${!var}" ]]; then
            missing_vars+=("$var")
        fi
    done
    
    if [[ ${#missing_vars[@]} -gt 0 ]]; then
        log_error "Missing required environment variables:"
        for var in "${missing_vars[@]}"; do
            echo "  - $var"
        done
        echo ""
        log_info "Please create a .env file with the required variables."
        log_info "See .env.example for reference."
        exit 1
    fi
    
    log_success "Environment validation passed"
}

# Create backup
create_backup() {
    if [[ "$ENVIRONMENT" == "production" ]]; then
        log_info "Creating backup..."
        mkdir -p "$BACKUP_DIR"
        
        # Backup database (if using local PostgreSQL)
        if command -v pg_dump &> /dev/null; then
            log_info "Backing up database..."
            pg_dump "$DATABASE_URL" > "$BACKUP_DIR/database.sql" 2>/dev/null || log_warning "Database backup skipped (using Supabase)"
        fi
        
        # Backup configuration files
        cp .env "$BACKUP_DIR/" 2>/dev/null || log_warning "No .env file to backup"
        cp docker-compose.yml "$BACKUP_DIR/" 2>/dev/null || true
        
        log_success "Backup created at $BACKUP_DIR"
    fi
}

# Run database migrations
run_migrations() {
    log_info "Running database migrations..."
    
    # Check if Supabase CLI is available
    if command -v supabase &> /dev/null; then
        log_info "Running Supabase migrations..."
        supabase db reset --local || log_warning "Local Supabase not available, skipping migrations"
    else
        log_warning "Supabase CLI not found. Please run database migrations manually."
        log_info "Execute the SQL in backend/database/schema.sql in your Supabase dashboard."
    fi
}

# Build and start services
build_and_start() {
    log_info "Building and starting services..."
    
    # Stop existing services
    docker-compose down 2>/dev/null || true
    
    # Build services
    log_info "Building Docker images..."
    docker-compose build --no-cache
    
    # Start services
    log_info "Starting services..."
    docker-compose up -d
    
    log_success "Services started successfully"
}

# Health checks
run_health_checks() {
    log_info "Running health checks..."
    
    # Wait for services to start
    sleep 10
    
    # Check API health
    log_info "Checking API health..."
    for i in {1..30}; do
        if curl -f http://localhost:8000/health >/dev/null 2>&1; then
            log_success "API health check passed"
            break
        elif [[ $i -eq 30 ]]; then
            log_error "API health check failed after 30 attempts"
            exit 1
        else
            log_info "Waiting for API to be ready... (attempt $i/30)"
            sleep 2
        fi
    done
    
    # Check frontend
    log_info "Checking frontend..."
    for i in {1..15}; do
        if curl -f http://localhost:3000 >/dev/null 2>&1; then
            log_success "Frontend health check passed"
            break
        elif [[ $i -eq 15 ]]; then
            log_error "Frontend health check failed after 15 attempts"
            exit 1
        else
            log_info "Waiting for frontend to be ready... (attempt $i/15)"
            sleep 2
        fi
    done
    
    # Check Redis
    log_info "Checking Redis..."
    if docker-compose exec -T redis redis-cli ping >/dev/null 2>&1; then
        log_success "Redis health check passed"
    else
        log_error "Redis health check failed"
        exit 1
    fi
}

# Main deployment process
main() {
    log_info "Starting ChainCraft deployment..."
    
    # Set up error handling
    trap cleanup EXIT
    
    # Load environment variables
    if [[ -f .env ]]; then
        log_info "Loading environment variables from .env"

}

user nginx;
worker_processes auto;
error_log /var/log/nginx/error.log notice;
pid /var/run/nginx.pid;

events {
    worker_connections 1024;
}

http {
    include /etc/nginx/mime.types;
    default_type application/octet-stream;

    log_format main '$remote_addr - $remote_user [$time_local] "$request" '
                    '$status $body_bytes_sent "$http_referer" '
                    '"$http_user_agent" "$http_x_forwarded_for"';

    access_log /var/log/nginx/access.log main;

    sendfile on;
    tcp_nopush on;
    keepalive_timeout 65;
    gzip on;
    gzip_types text/plain text/css application/json application/javascript text/xml application/xml application/xml+rss text/javascript;

    # Rate limiting
    limit_req_zone $binary_remote_addr zone=api:10m rate=10r/s;
    limit_req_zone $binary_remote_addr zone=frontend:10m rate=100r/s;

    # Upstream servers
    upstream api {
        server api:8000;
    }

    upstream frontend {
        server frontend:3000;
    }

    # HTTP to HTTPS redirect (for production)
    server {
        listen 80;
        server_name _;
        
        # Allow health checks
        location /health {
            proxy_pass http://api/health;
        }
        
        # Redirect all other traffic to HTTPS
        return 301 https://$server_name$request_uri;
    }

    # Main HTTPS server (for production)
    server {
        listen 443 ssl http2;
        server_name _;

        # SSL configuration
        ssl_certificate /etc/nginx/ssl/cert.pem;
        ssl_certificate_key /etc/nginx/ssl/key.pem;
        ssl_protocols TLSv1.2 TLSv1.3;
        ssl_ciphers HIGH:!aNULL:!MD5;
        ssl_prefer_server_ciphers on;

        # Security headers
        add_header X-Frame-Options DENY;
        add_header X-Content-Type-Options nosniff;
        add_header X-XSS-Protection "1; mode=block";
        add_header Strict-Transport-Security "max-age=31536000; includeSubDomains" always;

        # API routes
        location /api/ {
            limit_req zone=api burst=20 nodelay;
            
            proxy_pass http://api;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
            
            # WebSocket support
            proxy_http_version 1.1;
            proxy_set_header Upgrade $http_upgrade;
            proxy_set_header Connection "upgrade";
            
            # Timeouts
            proxy_connect_timeout 60s;
            proxy_send_timeout 60s;
            proxy_read_timeout 60s;
        }

        # Health check (no rate limiting)
        location /health {
            proxy_pass http://api/health;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        }

        # Frontend
        location / {
            limit_req zone=frontend burst=50 nodelay;
            
            proxy_pass http://frontend;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
            
            # Handle client-side routing
            try_files $uri $uri/ @frontend;
        }

        location @frontend {
            proxy_pass http://frontend;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        }

        # Static assets caching
        location ~* \.(js|css|png|jpg|jpeg|gif|ico|svg)$ {
            proxy_pass http://frontend;
            proxy_cache_valid 200 1d;
            add_header Cache-Control "public, max-age=86400";
        }
    }

    # Development server (HTTP only)
    server {
        listen 80 default_server;
        server_name localhost;

        # API routes
        location /api/ {
            limit_req zone=api burst=20 nodelay;
            
            proxy_pass http://api;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;

            proxy_set_header Upgrade $http_upgrade;
            proxy_set_header Connection "upgrade";

        # Health check

            proxy_pass http://api/health;
        }

        # Frontend

            limit_req zone=frontend burst=50 nodelay;

            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;

# /README.md
# ChainCraft Enterprise

A robust, scalable, and intuitive visual AI workflow builder that enables users to design, discover, and execute complex multi-agent AI chains.

## 🚀 Features

- **Visual Graph Editor**: Interactive canvas for building AI chains as interconnected nodes
- **AI Architect with RAG**: Intelligent assistant for conversational graph control
- **Multi-Provider Orchestration**: Support for OpenRouter, Runware, ElevenLabs, and more
- **Real-time Execution**: WebSocket-powered live updates during chain execution
- **Enterprise-grade Security**: Secure API key management and user data isolation
- **Cost Transparency**: Detailed cost tracking and estimation
- **Comprehensive Monitoring**: Performance tracking and error logging

## 🏗️ Architecture

### Backend (FastAPI + Python)
- **Core Services**: Graph execution engine, provider management, RAG system
- **Database**: Supabase (PostgreSQL) with Row Level Security
- **Task Queue**: Celery with Redis for background processing
- **Real-time**: WebSocket support for live updates
- **Monitoring**: Comprehensive logging and performance metrics

### Frontend (React.js)
- **Graph Editor**: React Flow integration for visual chain building
- **Chat Interface**: Conversational AI Architect interaction
- **Real-time Updates**: WebSocket client for execution progress
- **Theme System**: Customizable UI themes and fonts

### Infrastructure
- **Containerization**: Docker and Docker Compose
- **Reverse Proxy**: Nginx with SSL support
- **Scalability**: Horizontal scaling with load balancing

## 🛠️ Quick Start

### Prerequisites

- Docker & Docker Compose
- Node.js 18+ (for local development)
- Python 3.11+ (for local development)
- Supabase account
- API keys for providers (OpenRouter, Runware, ElevenLabs)

### 1. Clone and Setup

```bash
git clone <repository-url>
cd chaincraft-enterprise

# Run development setup script
./scripts/setup-dev.sh
```

### 2. Configure Environment

```bash
# Copy environment template
cp .env.example .env

# Edit .env with your credentials
nano .env
```

Required environment variables:
```bash
SUPABASE_URL=your_supabase_url
SUPABASE_SERVICE_KEY=your_service_role_key
SUPABASE_ANON_KEY=your_anon_key
OPENROUTER_API_KEY=your_openrouter_key
RUNWARE_API_KEY=your_runware_key
ELEVENLABS_API_KEY=your_elevenlabs_key
```

### 3. Setup Database

1. Go to your Supabase dashboard
2. Navigate to SQL Editor
3. Execute the SQL in `backend/database/schema.sql`

### 4. Deploy

```bash
# Full deployment
./deploy.sh

# Or start development environment
docker-compose up -d
```

### 5. Access Applications

- **Frontend**: http://localhost:3000
- **Backend API**: http://localhost:8000
- **API Documentation**: http://localhost:8000/docs
